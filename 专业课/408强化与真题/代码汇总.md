```cpp
bool bfs(int** G){

    //得到每个顶点的入度

    for(int i = 0; i < n; i ++)

        int v = q.front();
        q.pop();

        //得到了拓扑序的一个元素v，同时修改v指向的顶点的入度
        for(int i = 0; i < n; i ++)
            if(G[v][i]){
                d[i] --;
                if(d[i] == 0) q.push(i);
            }

        cnt ++;
    }

    if(cnt == n) return true;
    else return false;
}
```

### 拓扑排序（BFS实现）

基本思想是：

（1）一个数组维护每个顶点的入度，将初始度为0的顶点全部加入到队列中

（2）从入度为0的顶点开始bfs遍历，输出，同时更改它所有引出顶点的入度，如果这个点的入度为0，则加入队列成为后续bfs遍历队列的靠后的拓扑序元素

```cpp
int* bfs(int** G, int* result){
    int d[n];
    for(int i = 0; i < n; i ++) d[i] = 0;
    
    for(int i = 0; i < n; i ++)
        for(int j = 0; j < n; j ++)
            if(G[j][i]) d[i] ++;
    
    queue<int> q;
    for(int i = 0; i < n ; i++)
        if(d[i] == 0) q.push(i);
    
    int k = 0;
    while(q.size()){
        int v = q.front();
        q.pop();
        result[k ++] = v;
        for(int i = 0; i < n; i ++)
            if(G[v][i] == 1){
                d[i] --;
                if(d[i] == 0) q.push(i)
            }
    }
    
    return reuslt;
}
```

邻接矩阵这里在得到每个顶点的度的时候时间复杂度是O(n^2)

在BFS顶点的时候时间复杂度也是O(n^2)

所以总的复杂度就是O(n^2)



但是如果是邻接表的存储方式，在得到每个顶点的入度的时候需要遍历整个邻接表的所有边和顶点所以时间复杂度是O(n + e)

在BFS的时候，每个顶点都会遍历一遍，同时在遍历某个顶点的时候需要更改它引出的边的所有顶点的度，所以BFS的时候复杂度同样是O(n + e)

所以邻接表存储矩阵的话，拓扑排序的复杂度就是O(n + e)的级别



### 拓扑排序（DFS实现）

基本思想：

（1）dfs遍历这个顶点的所有引出顶点之后，再遍历这个顶点，并输出作为一个拓扑序元素

（2）这样对于任意A -> B这样一个顶点关系，A一定在B之后输出，最后的结果就是一个逆拓扑序，反转就是一个拓扑序了

有点像是树的后续遍历，将一个顶点的所有引出的顶点全部遍历完毕之后，返回的时候将这个顶点加入到拓扑

```cpp
int result[n];
vector<int> ans;
bool st[n];//dfs过程中遍历状态记录
int du[n];//记录所有顶点的入度
int G[n][n];//邻接矩阵
void dfs(int v){
    
    for(int i = 0; i < n; i ++){
        if(st[i] == 0 && G[v][i]){
            st[i] = 1;
            dfs(i);
        }
    }
    ans.push_back(v);
}

int main(){
    
    for(int i = 0; i < n; i ++) {
        st[i] = 0;
        du[i] = 0;
    }
    for(int i = 0; i < n; i ++)
        for(int j = 0; j < n; j ++)
            if(G[j][i]) du[i] ++;
    //为了防止不是一个连通图，必须从所有的入度为0的点开始
    for(int i = 0; i < n; i ++) if(du[i] == 0 && st[i] == 0) {
        st[i] == true;
        dfs(i);
    }
    ans.reverse();//将得到的逆拓扑序反序即可
    return 0;
}
```

### 二分查找（Acwing 模板）

找到比当前元素大的最小元素，因此是区间右侧性质的左端点，所以用的二分模板是下面：

```cpp
bool check(int mid){
    ......
}

void bsearch(int l, int r, int t){
    
    while(l < r){
    	int mid = l + r >> 1;
        //找到区间中右侧性质的右端点，check函数必须满足右侧性质，本题中就是>t
        if(check(mid)) r = mid;
    	else l = mid + 1;//mid不满足右侧性质，那么显然mid指向的是左边性质区域，我们应该向右缩小区域
    }
    return l;
}
```

如果我们要找的是左侧性质的右端点，则使用的模板是下面，并且 check () 函数需要满足左边区间的性质：
```cpp
bool check(){
}

void bsearch(int l, int r, int t){
	while(l < r){
		int mid = l + r + 1 >> 1;
		//找到左边区间的右端点，这里的check满足的是左边区间的性质并且改变的是左端点，显然此时的mid满足左边区间的形式，那么左端点就应该设置为mid，表示我们寻找的左侧性质的右端点在右边[mid, r]部分
		if(check(mid)) l = mid;
		else r = mid - 1//不满足左侧性质，那么此时mid肯定指向右侧性质，所以我们应该将区域往左缩小
	}
}
```

这样记忆：
（1）找左边区间右端点则 check 函数必须满足左侧性质；找右侧区间左端点，check 函数必须满足左侧性质，这里的满足性质指的是 check 函数中 mid 满足某侧性质后返回 ture，满足左侧性质就是 mid 满足左侧性质后返回 ture
（2）check 函数满足的是哪边的性质，那么满足 check 函数更改的就是哪边的端点；这时根据代码意思写出 if 和 else 两个语句即可
（3）mid = l + r + 1 >> 1 和某个区间 - 1 是绑定的，mid = l + r >> 1 与+1 是绑定的，先写出 check 函数，然后判断区域应该往哪里缩小，写出 else 后面的语句，然后再写 mid 应该如何改变

```cpp
void shell_sort(){
    //首先进行公差枚举
    for(int d = n / 2; d; d /= 2){
        //然后枚举每一组序列的起点
        //每组序列的起点都不会大于公差

            //对于每一组进行插入排序

            //还是从每组第二个元素开始

                //如果前一个元素大于后一个元素，则需要进行后移

                }
                //结束循环之后j就指向元素t应该放入的位置
                q[j] = t;
            }
        }
```

### 归并排序

```cpp
int q[n];//这里是待排的数组元素
int w[n];//这里是归并时候需要用到的辅助数组

void merge_sort(int l, int r){
    //如果l >= r说明此时递归到了只有一个元素，不用合并，直接返回即可
    if(l >= r) return;
    //首先确定划分中点
    int mid = l + r >> 1;
    //然后分别递归左右两边
    merge_sort(l, mid);
    merge_sort(mid + 1, r);//跟二分模板一样，如果mid = l + r >> 1则划分的区间是[l, mid], [mid + 1, r];
    //左右两边排好序之后进行二路归并
    int i = l;//左指针，从左边区间的第一个元素开始
    int j = mid + 1;//右指针，从右边区间的第一个元素开始
    int k = 0;//这里的k是临时数组的下标
    
    //如果左边没有走完并且右边没有走完的时候进行合并
    while(i <= mid && j <= r)
        if(q[i] <= q[j]) w[k ++] = q[i ++];//这里左边小于等于的时候将这个元素放入到辅助数组中，保持稳定性
    //这里注意应该加上等号
    	else w[k ++] = q[j ++];
    //上述循环结束的时候必然是一部分走完了，然后另一部分剩下了
    //此时剩下部分直接接到辅助数组的尾部即可
    while(i <= mid) w[k ++] = q[i ++];
    while(j <= r) w[k ++] = q[j ++];
    
    //接着将临时数组的元素复制到原来的数组中，此时合并完成
    //注意我们临时数组合并的其实是q[l ~ r]部分，因此q的下标应该从l开始，然后将临时数组中的k个数据赋值回q的[l, r]部分
    for(int i = l, j = 0; j < k; j ++, i ++) q[i] = w[j];
    
}
```

### TestAndSet指令和使用

```cpp
bool TestAndSet(bool* lock){
    bool old = *lock;
    *lock = true;
    return old;
}

//使用
while(TestAndSet(lock));
进入临界区
*lock = false;

```

（1）lock为所有进程共享，每个进程都可以操控lock变量，所以输入参数是一个指针变量指向lock

（2）TLS的代码作用就是返回lock旧值，同时将指针指向的lock设置为ture

（3）while起到阻拦的作用，lock = true就一直阻拦，直到别的进程推出临界区将lock设置成false，此时当下进程返回的旧值就是false，通过while阻拦进入临界区

### swap指令和使用

```cpp
void swap(bool* a, bool* b){

    temp = *a;
    *a = *b;
    *b = temp;
}

//swap指令的使用
bool* old = true;
while(old == true){
    swap(old, lock);
}
进入临界区;
*lock = false;
```

（1）swap指令本身只是用来实现两个bool指针指向的变量数值互换

（2）swap指令本质上和TSL指令是一样的，将lock赋值为true并返回lock的值用于while判断

（3）当别的进程在临界区中的时候由于old = true，所以lock被设置成true，当下进程被卡死在while循环中

（4）当别的进程出临界区，将lock设置为false，当下进程的old才会获得lock的新值false，才会出while循环
